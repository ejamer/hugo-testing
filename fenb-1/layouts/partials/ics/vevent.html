{{- /* Returns one VEVENT block (folded, CRLF-terminated) for an events.yaml entry, in the current site language. Shared by the season feed (layouts/events/list.calendar.ics) and the per-event download files (partials/ics/event-url.html). - All-day events: DTSTART/DTEND use VALUE=DATE, so there are no time zones to get wrong. DTEND is exclusive per RFC 5545, hence end date + 1 day. - UID is keyed off start_date + the English title, so it's the same in the EN and FR files — a calendar that imports both treats them as one event. Call with: {{ partial "ics/vevent.html" $event }} */ -}} {{- $e := . -}} {{- $fr := eq site.Language.Lang "fr" -}} {{- $pick := dict "title" (cond (and $fr $e.title_fr) $e.title_fr $e.title) "desc" (cond (and $fr $e.description_fr) $e.description_fr $e.description_en) "details" (cond (and $fr $e.details_url_fr) $e.details_url_fr $e.details_url_en) "register" (cond (and $fr $e.registration_url_fr) $e.registration_url_fr $e.registration_url_en) -}} {{- $start := time $e.start_date -}} {{- $end := $start -}} {{- with $e.end_date }}{{ $end = time . }}{{ end -}} {{- $descParts := slice -}} {{- with $pick.desc }}{{ $descParts = $descParts | append . }}{{ end -}} {{- with $pick.details }}{{ $descParts = $descParts | append (printf "%s: %s" (i18n "event_details") .) }}{{ end -}} {{- with $pick.register }}{{ $descParts = $descParts | append (printf "%s: %s" (i18n "event_register") .) }}{{ end -}} {{- $lines := slice "BEGIN:VEVENT" (printf "UID:%s-%s@fenb.ca" ($start.Format "20060102") (md5 $e.title)) (printf "DTSTAMP:%s" (now.UTC.Format "20060102T150405Z")) (printf "DTSTART;VALUE=DATE:%s" ($start.Format "20060102")) (printf "DTEND;VALUE=DATE:%s" (($end.AddDate 0 0 1).Format "20060102")) (printf "SUMMARY:%s" (partial "ics/escape.html" $pick.title)) (printf "LOCATION:%s" (partial "ics/escape.html" $e.location)) (printf "CATEGORIES:%s" (partial "ics/escape.html" (i18n $e.category | default $e.category))) -}} {{- with $descParts }} {{- $lines = $lines | append (printf "DESCRIPTION:%s" (partial "ics/escape.html" (delimit . "\n\n"))) -}} {{- end -}} {{- with $pick.details }}{{ if hasPrefix . "http" }} {{- $lines = $lines | append (printf "URL:%s" .) -}} {{- end }}{{ end -}} {{- $lines = $lines | append "END:VEVENT" -}} {{- $out := "" -}} {{- range $lines }}{{ $out = printf "%s%s" $out (partial "ics/line.html" .) }}{{ end -}} {{- return $out -}}
